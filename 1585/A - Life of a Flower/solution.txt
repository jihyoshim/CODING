t = int(input())
for _ in range(t) : 
  n = int(input())
  a = list(map(int, input().split()))
  height = 1
  is_dead = False 
  
  for i in range(n) : 
    if a[i] == 1 : 
      if i > 0 and a[i-1] == 1 : 
        height += 5
      else : 
        height += 1 
    else : 
      if i > 0 and a[i-1] == 0 : 
        is_dead = True
        break 
  if is_dead : 
    print(-1) 
  else : 
    print(height)
